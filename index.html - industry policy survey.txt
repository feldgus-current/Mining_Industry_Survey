        "text": "Permanently authorize MSP/FORGE (that is, the Mineral Security Project from the Biden Administration, which is being rebranded as FORGE under this administration).",
        "n": 66
      }
    ]
  }
];

const BUCKET_TOTAL = SECTIONS.reduce((a, s) => a + s.items.filter((it) => it.type === 'bucket').length, 0);
const OPEN_TOTAL = SECTIONS.reduce((a, s) => a + s.items.filter((it) => it.type === 'open').length, 0);
const TOTAL = BUCKET_TOTAL + OPEN_TOTAL;

function esc(s) {
  return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;")
                  .replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}

function build() {
  let html = "";
  SECTIONS.forEach(function (sec, si) {
    html += '<div class="qblock">';
    html += '<div class="qlabel">Section ' + (si + 1) + ' of ' + SECTIONS.length + '</div>';
    html += '<div class="qtext">' + esc(sec.title) + '</div>';

    const hasBucketItems = sec.items.some(function (it) { return it.type === 'bucket'; });
    if (hasBucketItems) {
      html += '<div class="grid-head"><span>Proposal</span>';
      OPTIONS.forEach(function (o) { html += '<span>' + esc(o) + '</span>'; });
      html += '<span>Comments</span></div>';
    }

    sec.items.forEach(function (it) {
      if (it.type === 'open') {
        html += '<div class="orow" id="row-' + it.n + '">'
              + '<div class="otext"><span class="pnum">' + it.n + '.</span>' + esc(it.text) + '</div>'
              + '<textarea class="oanswer" data-open="' + it.n + '" rows="2" placeholder="Your response"></textarea>'
              + '</div>';
        return;
      }
      html += '<div class="prow" id="row-' + it.n + '">';
      html += '<div class="ptext"><span class="pnum">' + it.n + '.</span>' + esc(it.text) + '</div>';
      OPTIONS.forEach(function (o, oi) {
        const id = "c-" + it.n + "-" + oi;
        html += '<div class="cellbox">'
              + '<input type="checkbox" id="' + id + '" data-item="' + it.n + '" value="' + esc(o) + '">'
              + '<label for="' + id + '">' + esc(o) + '</label></div>';
      });
      html += '<div><input type="text" class="pcomment" data-comment="' + it.n + '" placeholder="Optional"></div>';
      html += '</div>';
    });

    html += '<div class="secq-label">' + esc(SECTION_QUESTION) + '</div>';
    html += '<textarea class="qta" data-section="' + esc(sec.title) + '"></textarea>';
    html += '</div>';
  });

  document.getElementById("sections").innerHTML = html;
}

function collect() {
  const answers = {}, comments = {}, openAnswers = {}, sectionComments = {};

  document.querySelectorAll("input[type=checkbox][data-item]").forEach(function (cb) {
    if (cb.checked) {
      const k = cb.getAttribute("data-item");
      (answers[k] = answers[k] || []).push(cb.value);
    }
  });
  document.querySelectorAll("input[data-comment]").forEach(function (i) {
    const v = i.value.trim();
    if (v) comments[i.getAttribute("data-comment")] = v;
  });
  document.querySelectorAll("textarea[data-open]").forEach(function (t) {
    const v = t.value.trim();
    if (v) openAnswers[t.getAttribute("data-open")] = v;
  });
  document.querySelectorAll("textarea[data-section]").forEach(function (t) {
    const v = t.value.trim();
    if (v) sectionComments[t.getAttribute("data-section")] = v;
  });

  return {
    respondentId: respondentId(),
    respondent: document.getElementById("rname").value.trim(),
    organization: document.getElementById("rorg").value.trim(),
    email: document.getElementById("remail").value.trim(),
    answers: answers,
    comments: comments,
    openAnswers: openAnswers,
    sectionComments: sectionComments
  };
}

function refresh() {
  const d = collect();
  const bucketDone = Object.keys(d.answers).length;
  const openDone = Object.keys(d.openAnswers).length;
  const done = bucketDone + openDone;
  const pct = Math.round((done / TOTAL) * 100);
  document.getElementById("progress-text").textContent = done + " of " + TOTAL + " items answered";
  document.getElementById("progress-pct").textContent = pct + "%";
  document.getElementById("progress-fill").style.width = pct + "%";

  SECTIONS.forEach(function (sec) {
    sec.items.forEach(function (it) {
      const row = document.getElementById("row-" + it.n);
      if (!row) return;
      if (it.type === 'bucket') {
        row.classList.toggle("answered", !!d.answers[String(it.n)]);
      } else {
        row.classList.toggle("answered", !!d.openAnswers[String(it.n)]);
      }
    });
  });

  document.querySelectorAll(".cellbox").forEach(function (c) {
    const cb = c.querySelector("input");
    if (cb) c.classList.toggle("on", cb.checked);
  });

  try { localStorage.setItem(DRAFT_KEY, JSON.stringify(d)); } catch (e) {}
}

function restore() {
  let raw = null;
  try { raw = localStorage.getItem(DRAFT_KEY); } catch (e) { return; }
  if (!raw) return;
  let d;
  try { d = JSON.parse(raw); } catch (e) { return; }

  if (d.respondent) document.getElementById("rname").value = d.respondent;
  if (d.organization) document.getElementById("rorg").value = d.organization;
  if (d.email) document.getElementById("remail").value = d.email;

  Object.keys(d.answers || {}).forEach(function (k) {
    d.answers[k].forEach(function (v) {
      document.querySelectorAll('input[data-item="' + k + '"]').forEach(function (cb) {
        if (cb.value === v) cb.checked = true;
      });
    });
  });
  Object.keys(d.comments || {}).forEach(function (k) {
    const i = document.querySelector('input[data-comment="' + k + '"]');
    if (i) i.value = d.comments[k];
  });
  Object.keys(d.openAnswers || {}).forEach(function (k) {
    const t = document.querySelector('textarea[data-open="' + k + '"]');
    if (t) t.value = d.openAnswers[k];
  });
  document.querySelectorAll("textarea[data-section]").forEach(function (t) {
    const v = (d.sectionComments || {})[t.getAttribute("data-section")];
    if (v) t.value = v;
  });

  const status = document.getElementById("status");
  if (Object.keys(d.answers || {}).length || Object.keys(d.openAnswers || {}).length) {
    status.textContent = "Draft restored";
    status.className = "status ok";
  }
}

function clearAll() {
  if (!confirm("Clear every answer on this page?")) return;
  document.querySelectorAll("input[type=checkbox][data-item]").forEach(function (cb) { cb.checked = false; });
  document.querySelectorAll("input[data-comment]").forEach(function (i) { i.value = ""; });
  document.querySelectorAll("textarea[data-open]").forEach(function (t) { t.value = ""; });
  document.querySelectorAll("textarea[data-section]").forEach(function (t) { t.value = ""; });
  try { localStorage.removeItem(DRAFT_KEY); } catch (e) {}
  document.getElementById("status").textContent = "";
  refresh();
  window.scrollTo({ top: 0, behavior: "smooth" });
}

async function submit() {
  const d = collect();
  const statusEl = document.getElementById("status");

  if (!d.respondent || !d.organization) {
    statusEl.textContent = "Name and organization are required.";
    statusEl.className = "status err";
    window.scrollTo({ top: 0, behavior: "smooth" });
    return;
  }

  const payload = {
    respondentId: d.respondentId,
    respondent: d.respondent,
    organization: d.organization,
    email: d.email,
    items: [],
    sectionComments: []
  };

  SECTIONS.forEach(function (sec) {
    sec.items.forEach(function (it) {
      if (it.type === 'open') {
        payload.items.push({
          n: it.n,
          type: 'open',
          section: sec.title,
          proposal: it.text,
          buckets: [],
          comment: d.openAnswers[String(it.n)] || ""
        });
      } else {
        const picked = d.answers[String(it.n)] || [];
        payload.items.push({
          n: it.n,
          type: 'bucket',
          section: sec.title,
          proposal: it.text,
          buckets: picked,
          comment: d.comments[String(it.n)] || ""
        });
      }
    });
    const sc = d.sectionComments[sec.title];
    if (sc) payload.sectionComments.push({ section: sec.title, text: sc });
  });

  const btn = document.getElementById("btn-submit");
  btn.disabled = true;
  btn.textContent = "Submitting\u2026";
  statusEl.textContent = "";
  statusEl.className = "status";

  try {
    const res = await fetch(BACKEND, {
      method: "POST",
      headers: { "Content-Type": "text/plain;charset=utf-8" },
      body: JSON.stringify(payload)
    });
    const out = await res.json();
    if (out.status !== "ok") throw new Error(out.message || "Server rejected the submission");

    btn.textContent = "Submit responses";
    btn.disabled = false;
    statusEl.textContent = "Response recorded \u00b7 " + (out.id || "");
    statusEl.className = "status ok";
    const banner = document.getElementById("success-banner");
    banner.style.display = "block";
    banner.scrollIntoView({ behavior: "smooth", block: "nearest" });
  } catch (err) {
    btn.disabled = false;
    btn.textContent = "Submit responses";
    statusEl.textContent = "Submission failed, your draft is still saved. " + err.message;
    statusEl.className = "status err";
  }
}

build();
respondentId();
restore();
refresh();
document.addEventListener("change", refresh);
document.addEventListener("input", refresh);
document.getElementById("btn-clear").addEventListener("click", clearAll);
document.getElementById("btn-submit").addEventListener("click", submit);
</script>
